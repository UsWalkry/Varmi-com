4️⃣ SSH KEY KURULUMU (Gelecek için)
------------------------------------------
Sunucuda:
  # Public key authentication aktif mi kontrol et
  cat /etc/ssh/sshd_config | grep PubkeyAuthentication
  
  # Windows public key ekle
  mkdir -p ~/.ssh
  chmod 700 ~/.ssh
  # Windows'tan public key'i kopyalayıp ~/.ssh/authorized_keys'e ekle
  chmod 600 ~/.ssh/authorized_keys
